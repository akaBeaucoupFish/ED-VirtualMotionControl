# ==============================================================================
# CHECK KEY BINDS!
# Left Control+LeftAlt+Space=Enter Camera Mode
# npenter=Toggle Free Cam
# np2=FreeCamToggleHUD
# ==============================================================================
BLOCK enter_camera_mode
KEYTAP Left Control+LeftAlt+Space
PAUSE 0.2
KEYTAP np2
PAUSE 0.2
KEYTAP npenter
PAUSE 0.2
END
# ==============================================================================


# ==============================================================================
BLOCK forza_initial_position
AXES -40 0 0 0 0 0 0
PAUSE 1.2
AXES 0 0 0 0 0 30 0
PAUSE 0.4
AXES 0 0 0 15 0 0 0
PAUSE 0.35
AXES 0 0 0 0 30 0 0
PAUSE 1
AXES 0 -50 0 0 0 0 0
PAUSE 1.6
AXES 0 0 -100 0 0 0 0
PAUSE 5
AXES 0 0 0 0 0 0 0
END
# ==============================================================================


# ==============================================================================
# Initialise
# ==============================================================================
CALL enter_camera_mode
CALL forza_initial_position
# Tilt down
AXES 0 0 0 0 -25 0 0
PAUSE 1.8
# Back up
AXES 0 0 0 0 0 0 0
# ==============================================================================
# CHECK KEY BINDS!
# np5=<FixCameraWorldToggle>
# ==============================================================================
KEYTAP np5
PAUSE 0.1
